341 template <typename T, typename Fn>
342 auto _update_core(Vec& grad, const T& beta, Fn&& cut_strategy) -> CutStatus {
343 // reuse _v as grad_t = M * g (pre-allocated, no allocation)
344 for (size_t i = 0; i != this->_n; ++i) {
345 auto s = 0.0;
346 for (size_t j = 0; j != this->_n; ++j) {
347 s += this->_mq(i, j) * grad[j];
348 }
349 this->_v[i] = s;
350 }
351
352 auto omega = 0.0;
353 for (size_t i = 0; i != this->_n; ++i) {
354 omega += this->_v[i] * grad[i];
355 }
356
357 auto maybe = this->_prologue(beta, omega, std::forward<Fn>(cut_strategy));
358 if (!maybe) {
359 grad = this->_v;
360 return CutStatus::NoEffect;
361 }
362 auto result = *maybe;
363 if (result.status != CutStatus::Success) {
364 return result.status;
365 }
366
367 // n (n+1) / 2 + n
368 const auto r = result.sigma / omega;
369 for (size_t i = 0; i != this->_n; ++i) {
370 const auto rQg = r * this->_v[i];
371 for (size_t j = 0; j != i; ++j) {
372 this->_mq(i, j) -= rQg * this->_v[j];
373 this->_mq(j, i) = this->_mq(i, j);
374 }
375 this->_mq(i, i) -= rQg * this->_v[i];
376 }
377
378 this->_epilogue(result, omega, grad, true);
379 return result.status;
380 }

402 template <typename T, typename Fn>
403 auto _update_stable_core(Vec& g, const T& beta, Fn&& cut_strategy) -> CutStatus {
404 // _scratch = w = L^{-1}g (forward substitution)
405 this->_scratch = g;
406 for (size_t j = 0; j != this->_n - 1; ++j) {
407 for (size_t i = j + 1; i != this->_n; ++i) {
408 this->_mq(j, i) = this->_mq(i, j) * this->_scratch[j];
409 this->_scratch[i] -= this->_mq(j, i);
410 }
411 }
412
413 // _z = D^{-1} * w
414 this->_z = this->_scratch;
415 for (size_t i = 0; i != this->_n; ++i) {
416 this->_z[i] *= this->_mq(i, i);
417 }
418
419 // omega = sum(w_i * z_i)
420 auto omega = 0.0;
421 for (size_t i = 0; i != this->_n; ++i) {
422 omega += this->_z[i] * this->_scratch[i];
423 }
424
425 auto maybe = this->_prologue(beta, omega, std::forward<Fn>(cut_strategy));
426 if (!maybe) {
427 return CutStatus::NoEffect;
428 }
429 auto result = *maybe;
430 if (result.status != CutStatus::Success) {
431 return result.status;
432 }
433
434 // _v = grad_t = L^{-T} * z (back substitution) — preserved for final g
435 this->_v = this->_z;
436 for (auto i = this->_n - 1; i != 0; --i) {
437 for (auto j = i; j != this->_n; ++j) {
438 this->_v[i - 1] -= this->_mq(j, i - 1) * this->_v[j];
439 }
440 }
441
442 // rank-one LDL^T update — reuse _scratch as working vector v
443 // (_scratch (w) no longer needed; overwrite with gradient)
444 const auto mu = result.sigma / (1.0 - result.sigma);
445 auto oldt = omega / mu;
446 this->_scratch = g; // v = gradient
447 for (size_t j = 0; j != this->_n; ++j) {
448 const auto p = this->_scratch[j];
449 const auto temp = this->_z[j];
450 const auto newt = oldt + p * temp;
451 const auto beta2 = temp / newt;
452 this->_mq(j, j) *= oldt / newt;
453 for (auto k = j + 1; k != this->_n; ++k) {
454 this->_scratch[k] -= this->_mq(j, k);
455 this->_mq(k, j) += beta2 * this->_scratch[k];
456 }
457 oldt = newt;
458 }
459 // _v still holds grad_t = L^{-T}*z (preserved from back substitution)
460 this->_epilogue(result, omega, g, false);
461 return result.status;
462 }
