285 template <typename T, typename Fn>
286 auto _update_core(Vec& grad, const T& beta, Fn&& cut_strategy) -> CutStatus {
287 // reuse _v as grad_t = M * g (pre-allocated, no allocation)
288 this->_v = 0.0;
289 for (size_t i = 0; i != this->_n; ++i) {
290 for (size_t j = 0; j != this->_n; ++j) {
291 this->_v[i] += this->_mq(i, j) * grad[j];
292 }
293 }
294
295 const auto omega = (this->_v * grad).sum();
296 this->_tsq = this->_kappa * omega;
297
298 if (omega <= std::numeric_limits<double>::min()) {
299 grad = this->_v;
300 return CutStatus::NoEffect;
301 }
302
303 auto result = std::forward<Fn>(cut_strategy)(beta, this->_tsq);
304 if (result.status != CutStatus::Success) {
305 return result.status;
306 }
307
308 // n (n+1) / 2 + n
309 const auto r = result.sigma / omega;
310 for (size_t i = 0; i != this->_n; ++i) {
311 const auto rQg = r * this->_v[i];
312 for (size_t j = 0; j != i; ++j) {
313 this->_mq(i, j) -= rQg * this->_v[j];
314 this->_mq(j, i) = this->_mq(i, j);
315 }
316 this->_mq(i, i) -= rQg * this->_v[i];
317 }
318
319 this->_kappa *= result.delta;
320
321 if (this->no_defer_trick) {
322 this->_mq *= this->_kappa;
323 this->_kappa = 1.0;
324 }
325
326 grad = this->_v * (result.rho / omega);
327 return result.status;
328 }

350 template <typename T, typename Fn>
351 auto _update_stable_core(Vec& g, const T& beta, Fn&& cut_strategy) -> CutStatus {
352 // _scratch = w = L^{-1}g (forward substitution)
353 this->_scratch = g;
354 for (size_t j = 0; j != this->_n - 1; ++j) {
355 for (size_t i = j + 1; i != this->_n; ++i) {
356 this->_mq(j, i) = this->_mq(i, j) * this->_scratch[j];
357 this->_scratch[i] -= this->_mq(j, i);
358 }
359 }
360
361 // _z = D^{-1} * w
362 this->_z = this->_scratch;
363 for (size_t i = 0; i != this->_n; ++i) {
364 this->_z[i] *= this->_mq(i, i);
365 }
366
367 // omega = sum(w_i * z_i)
368 auto omega = 0.0;
369 for (size_t i = 0; i != this->_n; ++i) {
370 omega += this->_z[i] * this->_scratch[i];
371 }
372
373 this->_tsq = this->_kappa * omega;
374
375 if (omega <= std::numeric_limits<double>::min()) {
376 return CutStatus::NoEffect;
377 }
378
379 auto result = std::forward<Fn>(cut_strategy)(beta, this->_tsq);
380 if (result.status != CutStatus::Success) {
381 return result.status;
382 }
383
384 // _v = grad_t = L^{-T} * z (back substitution) — preserved for final g
385 this->_v = this->_z;
386 for (auto i = this->_n - 1; i != 0; --i) {
387 for (auto j = i; j != this->_n; ++j) {
388 this->_v[i - 1] -= this->_mq(j, i - 1) * this->_v[j];
389 }
390 }
391
392 // rank-one LDL^T update — reuse _scratch as working vector v
393 // (_scratch (w) no longer needed; overwrite with gradient)
394 const auto mu = result.sigma / (1.0 - result.sigma);
395 auto oldt = omega / mu;
396 this->_scratch = g; // v = gradient
397 for (size_t j = 0; j != this->_n; ++j) {
398 const auto p = this->_scratch[j];
399 const auto temp = this->_z[j];
400 const auto newt = oldt + p * temp;
401 const auto beta2 = temp / newt;
402 this->_mq(j, j) *= oldt / newt;
403 for (auto k = j + 1; k != this->_n; ++k) {
404 this->_scratch[k] -= this->_mq(j, k);
405 this->_mq(k, j) += beta2 * this->_scratch[k];
406 }
407 oldt = newt;
408 }
409 this->_kappa *= result.delta;
410 // _v still holds grad_t = L^{-T}*z (preserved from back substitution)
411 g = this->_v * (result.rho / omega);
412 return result.status;
413 }
