261 auto _update_core(const std::pair<Arr, T>& cut, Fn&& cut_strategy) -> CutStatus {
262 const auto& grad = cut.first;
263 const auto& beta = cut.second;
264 auto& g = this->_g;
265 for (size_t i = 0; i != this->_n; ++i) {
266 g[i] = grad[i];
