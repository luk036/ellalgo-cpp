33inline auto stalled(double tsq, std::optional<double> tsq_prev) -> bool {

127inline auto cutting_plane_feas(O& omega, S& space, const Options& options = Options())
128 -> SolverResult<CuttingPlaneArrayType<S>> {
129 std::optional<double> tsq_prev;
130 for (auto niter = 0U; niter != options.max_iters; ++niter) {
131 const auto cut = omega.assess_feas(space.xc());
132 if (!cut) { // feasible sol'n obtained
133 return {space.xc(), niter, SolverStatus::Success};
134 }
135 const auto status = space.update_bias_cut(*cut); // update space
136 if (status != CutStatus::Success) {
137 auto res = invalid_value<CuttingPlaneArrayType<S>>();
138 return {std::move(res), niter, SolverStatus::Infeasible};
139 }
140 const auto tsq = space.tsq();
141 if (tsq < options.tolerance) {
142 auto res = invalid_value<CuttingPlaneArrayType<S>>();
143 return {std::move(res), niter, SolverStatus::Infeasible};
144 }
145 if (options.stall_guard && stalled(tsq, tsq_prev)) {
146 auto res = invalid_value<CuttingPlaneArrayType<S>>();
147 return {std::move(res), niter, SolverStatus::Stalled};
148 }
149 tsq_prev = tsq;
150 }
151 auto res = invalid_value<CuttingPlaneArrayType<S>>();
152 return {std::move(res), options.max_iters, SolverStatus::MaxIters};
153}

206template <typename O, typename S, typename N>
207 requires OracleOptim<O, typename S::ArrayType, N> && SearchSpace<S>
208inline auto cutting_plane_optim(O& omega, S& space, N& gamma, const Options& options = Options())
209 -> SolverResult<CuttingPlaneArrayType<S>> {
210 auto x_best = invalid_value<CuttingPlaneArrayType<S>>();
211 std::optional<double> tsq_prev;
212 for (auto niter = 0U; niter < options.max_iters; ++niter) {
213 const auto _result1 = omega.assess_optim(space.xc(), gamma);
214 const auto& cut = std::get<0>(_result1);
215 const auto& shrunk = std::get<1>(_result1);
216 const auto status = [&]() {
217 if (shrunk) { // best gamma obtained
218 x_best = space.xc();
219 return space.update_central_cut(cut); // should update_central_cut
220 }
221 return space.update_bias_cut(cut);
222 }();
223 if (status != CutStatus::Success) { // no more
224 const auto outcome
225 = x_best.size() != 0U ? SolverStatus::Success : SolverStatus::Infeasible;
226 return {std::move(x_best), niter, outcome};
227 }
228 const auto tsq = space.tsq();
229 if (tsq < options.tolerance) { // no more
230 const auto outcome
231 = x_best.size() != 0U ? SolverStatus::Success : SolverStatus::Infeasible;
232 return {std::move(x_best), niter, outcome};
233 }
234 if (options.stall_guard && stalled(tsq, tsq_prev)) {
235 const auto outcome
236 = x_best.size() != 0U ? SolverStatus::Success : SolverStatus::Stalled;
237 return {std::move(x_best), niter, outcome};
238 }
239 tsq_prev = tsq;
240 }
241 return {std::move(x_best), options.max_iters, SolverStatus::MaxIters};
242} // END

347template <typename O, typename S, typename N>
348 requires OracleOptimQ<O, typename S::ArrayType, N> && SearchSpace<S>
349inline auto cutting_plane_optim_q(O& omega, S& space_q, N& gamma,
350 const Options& options = Options())
351 -> SolverResult<CuttingPlaneArrayType<S>> {
352 using A = CuttingPlaneArrayType<S>;
353 OptimQState<A> state{invalid_value<A>()};
354 std::optional<double> tsq_prev;
355
356 for (auto niter = 0U; niter < options.max_iters; ++niter) {
357 const auto result1 = omega.assess_optim_q(space_q.xc(), gamma, state.retry());
358 const auto& cut = std::get<0>(result1);
359 const auto& shrunk = std::get<1>(result1);
360 if (shrunk) { // best gamma obtained
361 state.on_shrunk(std::move(std::get<2>(result1)));
362 }
363 const auto outcome = state.on_update(space_q.update_q(cut), std::get<3>(result1));
364 if (outcome == OptimQState<A>::Result::NoSoln) {
365 const auto res
366 = state.x_best().size() != 0U ? SolverStatus::Success : SolverStatus::Infeasible;
367 return {std::move(state.x_best()), niter, res};
368 }
369 if (outcome == OptimQState<A>::Result::NoMoreAlt) {
370 break; // no more alternative cut
371 }
372 const auto tsq = space_q.tsq();
373 if (tsq < options.tolerance) { // no more
374 const auto res
375 = state.x_best().size() != 0U ? SolverStatus::Success : SolverStatus::Infeasible;
376 return {std::move(state.x_best()), niter, res};
377 }
378 if (options.stall_guard && stalled(tsq, tsq_prev)) {
379 const auto res
380 = state.x_best().size() != 0U ? SolverStatus::Success : SolverStatus::Stalled;
381 return {std::move(state.x_best()), niter, res};
382 }
383 tsq_prev = tsq;
384 }
385 return {std::move(state.x_best()), options.max_iters, SolverStatus::MaxIters};
386} // END

472 requires OracleBS<O, T>
473inline auto bsearch(O& omega, const std::pair<T, T>& intvl, const Options& options = Options())
474 -> SolverResult<T> {
475 // assume monotone
476 // auto& [lower, upper] = intvl;
477 auto lower = intvl.first;
478 auto upper = intvl.second;
479 assert(lower <= upper);
480
481 for (auto niter = 0U; niter < options.max_iters; ++niter) {
482 auto tau = algo::half_nonnegative(upper - lower);
483 if (tau < options.tolerance) { // no more
484 return {upper, niter, SolverStatus::Success};
485 }
486 auto gamma = lower; // l may be `int` or `Fraction`
487 gamma += tau;
488 // The midpoint stops moving once the bracket reaches floating-point
489 // resolution. `tau < options.tolerance` is then unreachable for any
490 // realistic scale (tolerance defaults to 1e-20, while the interval
491 // underflows at ~1e-16 of its own magnitude), so without this guard the
492 // loop would spin until max_iters without refining anything.
493 if (!(lower < gamma && gamma < upper)) {
494 return {upper, niter, SolverStatus::Success};
495 }
496 if (omega.assess_bs(gamma)) { // feasible sol'n obtained
497 upper = gamma;
498 } else {
499 lower = gamma;
500 }
501 }
502 return {upper, options.max_iters, SolverStatus::MaxIters};
503}
