15// Modified from CVX code by Almir Mutapcic in 2006.
16// Adapted in 2010 for impulse response peak-minimization by convex iteration by
17// Christine Law.
18//
19// "FIR Filter Design via Spectral Factorization and Convex Optimization"
20// by S.-P. Wu, S. Boyd, and L. Vandenberghe
21//
22// Designs an FIR lowpass filter using spectral factorization method with
23// constraint on maximum passband ripple and stopband attenuation:
24//
25// minimize max |H(w)| for w in stopband
26// s.t. 1/delta <= |H(w)| <= delta for w in passband
27//
28// We change variables via spectral factorization method and get:
29//
30// minimize max R(w) for w in stopband
31// s.t. (1/delta)**2 <= R(w) <= delta**2 for w in passband
32// R(w) >= 0 for all w
33//
34// where R(w) is squared magnitude frequency response
35// (and Fourier transform of autocorrelation coefficients r).
36// Variables are coeffients r and G = hh' where h is impulse response.
37// delta is allowed passband ripple.
38// This is a convex problem (can be formulated as an SDP after sampling).

153inline auto create_lowpass_case(size_t N = 48) -> std::pair<LowpassOracle, double> {
154 const auto delta0_wpass = 0.025;
155 const auto delta0_wstop = 0.125;
156 // maximum passband ripple in dB (+/- around 0 dB)
157 const auto delta1 = 20.0 * std::log10(1.0 + delta0_wpass);
158 // stopband attenuation desired in dB
159 const auto delta2 = 20.0 * std::log10(delta0_wstop);
160
161 // *********************************************************************
162 // optimization parameters
163 // *********************************************************************
164 // rule-of-thumb discretization (from Cheney's Approximation Theory)
165 const auto Lp = std::pow(10, -delta1 / 20);
166 const auto Up = std::pow(10, +delta1 / 20);
167 const auto Sp = std::pow(10, +delta2 / 20);
168
169 auto omega = LowpassOracle(N, Lp * Lp, Up * Up, 0.12, 0.20);
170 return {std::move(omega), Sp * Sp};
171}
